mps4_board.subsystem.ethosu.num_macs=256              # (int   , init-time) default = '0x400'  : Number of 8x8 MACs performed per cycle (128, 256, 512, 1024, or 2048). : [0x80:0x800]
                                                      # NOTE: must match the Vela target the model was compiled for:
                                                      # --accelerator-config ethos-u85-256 in cmsis-executorch.cbuild-mlops.yml
                                                      # (from the mlops: node of the csolution), otherwise the command
                                                      # stream is rejected at runtime ("npu.macs_per_cc != optimizer.macs_per_cc").
mps4_board.subsystem.cpu0.INITSVTOR=0x12000000        # (int   , init-time) default = '0x11000000' : Secure vector-table offset at reset
mps4_board.subsystem.iotss3_systemcontrol.INITSVTOR_RST=0x12000000  # (int, init-time) default = '0x11000000' : Reset value for INITSVTOR. Should match cpu0.INITSVTOR
                                                      # NOTE: the application's vector table (__Vectors) is linked to FPGA SRAM at
                                                      # 0x12000000 (__ROM0_BASE in regions_SSE-320.h), but the FVP default points
                                                      # at 0x11000000 where nothing is loaded. The first boot works anyway
                                                      # because the ELF loader sets the PC directly -- but any debugger reset
                                                      # reboots from INITSVTOR, fetches a garbage MSP/PC and locks the core up
                                                      # (HardFault escalation during exception entry). Pin both to the real
                                                      # vector table so resets survive.
