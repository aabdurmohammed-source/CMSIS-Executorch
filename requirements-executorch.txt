# ExecuTorch and torchao. Separate from requirements.txt because these resolve
# from the PyTorch nightly index, and pointing pip at that index for the whole
# install would let it prefer a nightly `torch` build over the PyPI release
# pinned in requirements.txt. setup_venv.py installs this file on its own.
#
# ExecuTorch 1.4.0 is not on PyPI yet (the newest published release is 1.3.1).
# These wheels are cut from the same release/1.4 branch as the
# PyTorch::ExecuTorch@1.4.0 pack, and are prebuilt for Linux, macOS and Windows.
# Once 1.4.0 is on PyPI, `executorch==1.4.0` can move into requirements.txt.
--index-url https://download.pytorch.org/whl/nightly/cpu
--extra-index-url https://pypi.org/simple
executorch==1.4.0.dev20260728

# The Arm quantizer imports torchao. ExecuTorch 1.4 pins the nightly below
# (TORCHAO_NIGHTLY_VERSION in install_requirements.py); PyPI is still on 0.17.0,
# so this one cannot come from requirements.txt yet.
torchao==0.18.0.dev20260715
